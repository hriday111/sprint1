605int sweep_tui_run(disease_assign_t *assign, const sweep_tui_hooks_t *hooks)
606{
607 ui_t ui;
608 int last_cols;
609 int last_rows;
610
611 memset(&g, 0, sizeof(g));
612 g.assign = assign;
613 if (hooks)
614 g.hooks = *hooks;
615 g.phase = PICK_BLOB1;
616 assign->blob1 = 0;
617 assign->blob2 = 0;
618
619 ui_new(SCR_PICK, &ui);
620 ui.canscroll = 0;
621 g.ui = &ui;
622 g.panel_w = panel_width(&ui);
623 last_cols = ui.ws.ws_col;
624 last_rows = ui.ws.ws_row;
625
626 set_status(ST_READY, "Pick diseases first.", "Choose D1–D10 for Blob 1.");
627 build_all(&ui);
628 ui.screen = SCR_PICK;
629 ui_draw(&ui);
630
631 {
