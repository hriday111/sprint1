200int vec_expand_(char **data, int *length, int *capacity, int memsz) {
201 if (*length + 1 > *capacity) {
202 void *ptr;
203 int n = (*capacity == 0) ? 1 : *capacity << 1;
204 ptr = realloc(*data, n * memsz);
205 if (ptr == NULL) return -1;
206 *data = ptr;
207 *capacity = n;
208 }
209 return 0;
210}

213int vec_reserve_(char **data, int *length, int *capacity, int memsz, int n) {
214 (void) length;
215 if (n > *capacity) {
216 void *ptr = realloc(*data, n * memsz);
217 if (ptr == NULL) return -1;
218 *data = ptr;
219 *capacity = n;
220 }
221 return 0;
222}

225int vec_reserve_po2_(
226 char **data, int *length, int *capacity, int memsz, int n
227) {
228 int n2 = 1;
229 if (n == 0) return 0;
230 while (n2 < n) n2 <<= 1;
231 return vec_reserve_(data, length, capacity, memsz, n2);
232}

235int vec_compact_(char **data, int *length, int *capacity, int memsz) {
236 if (*length == 0) {
237 free(*data);
238 *data = NULL;
239 *capacity = 0;
240 return 0;
241 } else {
242 void *ptr;
243 int n = *length;
244 ptr = realloc(*data, n * memsz);
245 if (ptr == NULL) return -1;
246 *capacity = n;
247 *data = ptr;
248 }
249 return 0;

648void _ui_update(char *c, int n, ui_t *u){

675 case '6':

683 }

694/*
695 * Visible (cell) length of an ANSI-colored string, ignoring escape codes.
696 */
697int ui_visible_len(char *str){
698 struct vt100_node_t *head, *tmp;
699 int n = 0;
700
701 if(str == NULL || *str == '\0') return 0;
702
703 head = vt100_decode(str);
704 for(tmp = head; tmp != NULL; tmp = tmp->next){
705 if(tmp->str == NULL || tmp->str[0] == '\0') continue;
706 /* vt100_decode stores strlen+1 in len when it copies text */
707 n += (tmp->len > 1) ? (tmp->len - 1) : (int)strlen(tmp->str);
708 }
709 vt100_free(head);
710 return n;
711}

713/*
714 * Wrap an ANSI-colored string to `width` terminal columns.
715 * Writes newline-separated lines into `out` (NUL-terminated).
716 * Returns the number of lines produced (at least 1 for non-empty input).
717 *
718 * Colors are preserved across wrap boundaries via vt100_sgr.
719 */
720int ui_wrap_text(char *str, int width, char *out, int outsz){
721 struct vt100_node_t *head, *tmp;
722 int x = 0, off = 0, len = 0, lines = 1;
723 char *sgr;
724 int vis, chunk;
725
726 if(outsz < 1) return 0;
727 out[0] = '\0';
728 if(str == NULL || *str == '\0') return 0;
729 if(width < 4) width = 4;
730
731 head = vt100_decode(str);
732 if(head == NULL) return 0;
733 tmp = head;
734
735 while(tmp != NULL){
736 if(tmp->str == NULL || tmp->str[0] == '\0'){
737 tmp = tmp->next;
738 continue;
739 }
740 vis = (tmp->len > 1) ? (tmp->len - 1) : (int)strlen(tmp->str);
741 off = 0;
742
743 while(off < vis){
744 if(x >= width){
745 if(len + 2 >= outsz) goto done;
746 out[len++] = '\n';
747 out[len] = '\0';
748 lines++;
749 x = 0;
750 }
751 chunk = width - x;
752 if(chunk > vis - off) chunk = vis - off;
753 if(chunk < 1) break;
754 if(len >= outsz - 1) goto done;
755
756 sgr = vt100_sgr(tmp, NULL);
757 len += snprintf(
758 out + len, (size_t)(outsz - len),
759 "%s%.*s", sgr, chunk, tmp->str + off
760 );
761 free(sgr);
762 if(len >= outsz - 1){
763 len = outsz - 1;
764 out[len] = '\0';
765 goto done;
766 }
767
768 off += chunk;
769 x += chunk;
770 }
771 tmp = tmp->next;
772 }
773
774 if(len + 5 < outsz)
775 len += sprintf(out + len, "\x1b[0m");
776
777done:
778 vt100_free(head);
779 return lines;
780}
