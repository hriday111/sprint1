91int roi_interp_at(const roi_interp_t *model, int pos, rois_t *out)
92{
93 memset(out, 0, sizeof(*out));
94 if (!model || model->at0.n < 1 || model->at0.n != model->at1.n)
95 return -1;
96 if (model->pos1 == model->pos0)
97 return -1;
98
99 double t = (double)(pos - model->pos0) / (double)(model->pos1 - model->pos0);
100 out->n = model->at0.n;
101 for (int i = 0; i < out->n; i++) {
102 const roi_t *a = &model->at0.item[i];
103 const roi_t *b = &model->at1.item[i];
104 roi_t *r = &out->item[i];
105 memset(r, 0, sizeof(*r));
106 snprintf(r->name, sizeof(r->name), "%s", a->name);
107 r->cx = a->cx + t * (b->cx - a->cx);
108 r->cy = a->cy + t * (b->cy - a->cy);
109 r->rx = a->rx + t * (b->rx - a->rx);
110 r->ry = a->ry + t * (b->ry - a->ry);
111 r->angle = a->angle + t * (b->angle - a->angle);
112 if (r->rx < 1.0)
113 r->rx = 1.0;
114 if (r->ry < 1.0)
115 r->ry = 1.0;
116 roi_sync_aabb(r);
117 }
118 return 0;
