177static void queue_goto(stepper_t *stepper, int pos)
178{
179 int diff;
180
181 if (!stepper->init) {
182 stepper->error = 1;
183 return;
184 }
185
186 diff = pos - stepper->pos;
187 if (diff > 0) {
188 stepper->dir = 1;
189 stepper->steps = diff;
190 } else if (diff < 0) {
191 stepper->dir = 0;
192 stepper->steps = -diff;
193 } else {
194 stepper->steps = 0;
195 }
196 stepper->error = 0;
197}

204static void *stepper_thread(void *arg)
205{
206 (void)arg;
207
208 while (g_thread_run) {
209 int do_step = 0;
210 int dir = 0;
211
212 pthread_mutex_lock(&g_mutex);
213 if (g_stepper != NULL && g_stepper->ena && g_stepper->steps > 0) {
214 do_step = 1;
215 dir = g_stepper->dir;
216 }
217 pthread_mutex_unlock(&g_mutex);
218
219 if (do_step) {
220 pulse_gpio(dir);
221
222 pthread_mutex_lock(&g_mutex);
223 if (g_stepper != NULL) {
224 account_step(g_stepper, dir);
225 if (g_stepper->steps > 0) {
226 g_stepper->steps--;
227 }
228 }
229 pthread_mutex_unlock(&g_mutex);
230 } else {
231 usleep(1000);
232 }
233 }
234
235 return NULL;
236}

323int stepperSend(int cmd, stepper_t *stepper)
324{
325 pthread_mutex_lock(&g_mutex);
326
327 if (cmd >= CMD_ANGLE_MIN && cmd <= CMD_ANGLE_MAX) {
328 queue_goto(stepper, cmd);
329 } else if (cmd == CMD_LASER_OFF) {
330 laser_set_unlocked(stepper, 0);
331 } else if (cmd == CMD_LASER_ON) {
332 laser_set_unlocked(stepper, LASER_PWM_RANGE);
333 } else if (cmd == CMD_DISABLE) {
334 disable_motor(stepper);
335 } else if (cmd == CMD_ENABLE) {
336 enable_motor(stepper);
337 } else if (cmd == CMD_RESET) {
338 /* Unreachable while CMD_RESET == CMD_DISABLE (both 1003). */
339 stepper->steps = 0;
340 stepper->error = 0;
341 } else if (cmd == CMD_NEXT) {
342 if (stepper->ena) {
343 pulse_step(stepper, 1);
344 stepper->dir = 1;
345 }
346 } else if (cmd == CMD_PREV) {
347 if (stepper->ena) {
348 pulse_step(stepper, 0);
349 stepper->dir = 0;
350 }
351 } else if (cmd == CMD_UPDATE_LID) {
352 update_lid(stepper);
353 } else {
354 LOG_WARN("stepperSend: unknown command %d", cmd);
355 }
356
357 pthread_mutex_unlock(&g_mutex);
358 return 1;
359}

446int stepperInit(stepper_t *stepper)
447{
448 int pos = -1;
449 int attempts = 0;
450
451 LOG_INFO("Homing: enable motor and seek endstop");
452 stepperSend(CMD_ENABLE, stepper);
453 ms_sleep(100);
454
455 while (pos < 0 && attempts < 2000) {
456 stepperSend(CMD_NEXT, stepper);
457 pos = stepperRead(stepper);
458 attempts++;
459 if ((attempts % 50) == 0) {
460 LOG_DEBUG("Homing... steps=%d endstop=%d", attempts,
461 digitalRead(PIN_ENDSTOP));
462 }
463 }
464 stepperSend(CMD_NEXT, stepper);
465 if (pos < 0) {
466 LOG_ERROR("Homing failed after %d steps (endstop never HIGH on GPIO %d)",
467 attempts, PIN_ENDSTOP);
468 return -1;
469 }
470
471 pthread_mutex_lock(&g_mutex);
472 stepper->zero = pos;
473 pthread_mutex_unlock(&g_mutex);
474
475 LOG_INFO("Endstop hit after %d steps, position=%d", attempts, pos);
476 return pos;
477}
