35static void loadParams(char *filename, stepper_t *stepper)
36{
37 int fd = open(filename, O_RDONLY);
38 char buffer[32];
39 int count;
40 int n;
41
42 if (fd == -1) {
43 LOG_ERROR("loadParams: open(%s) failed: %s", filename, strerror(errno));
44 ERR("open");
45 }
46
47 memset(buffer, 0, sizeof(buffer));
48 count = read(fd, buffer, sizeof(buffer) - 1);
49 close(fd);
50
51 if (count <= 0) {
52 LOG_FATAL("loadParams: empty or unreadable file %s", filename);
53 exit(EXIT_FAILURE);
54 }
55
56 buffer[count] = '\0';
57 n = sscanf(buffer, "%d %d %d", &stepper->start, &stepper->stop, &stepper->prep);
58 if (n != 3) {
59 LOG_FATAL("loadParams: expected 3 integers, got %d", n);
60 exit(EXIT_FAILURE);
61 }
62
63 /*
64 * Guard against nonsense / truncated reads. Keep in production until the
65 * loader is hardened (longer buffer, stricter parsing).
66 */
67 if (stepper->start < 0 || stepper->stop < 0 || stepper->prep < 0 ||
68 stepper->start > 999 || stepper->stop > 999 || stepper->prep > 999) {
69 LOG_FATAL("loadParams: unsupported values start=%d stop=%d prep=%d",
70 stepper->start, stepper->stop, stepper->prep);
71 exit(EXIT_FAILURE);
72 }
73 LOG_INFO("Params: start=%d stop=%d prep=%d", stepper->start, stepper->stop,
74 stepper->prep);
75}

140static int handle_command(char *line, stepper_t *stepper)
141{
142 char *end;
143 long value;
144
145 while (isspace((unsigned char)*line))
146 line++;
147 end = line + strlen(line);
148 while (end > line && isspace((unsigned char)end[-1]))
149 *--end = '\0';
150 if (*line == '\0')
151 return 0;

157 LOG_WARN("Ignoring bad brightness command: '%s'", line);
158 return 0;
159 }

162 return 0;
163 }
164
165 errno = 0;

169 return 0;
170 }
171
172 if (value == -1) {

174 return 1;

186 }

200static void serve_fifo(const char *path, stepper_t *stepper)
201{
202 char buf[128];
203 size_t len = 0;
204 int done = 0;
205
206 make_fifo(path);
207 make_fifo(POS_PIPE);
208 LOG_INFO("Serving commands on %s", path);
209
210 while (!done) {
211 int fifo = open(path, O_RDONLY);
212 if (fifo < 0) {
213 LOG_ERROR("Open fifo failed (%s), retrying", strerror(errno));
214 usleep(100000);
215 continue;
216 }
217 LOG_DEBUG("Writer connected on command FIFO");
218
219 for (;;) {
220 ssize_t count = read(fifo, buf + len, sizeof(buf) - 1 - len);
221 if (count < 0) {
222 if (errno == EINTR)
223 continue;
224 ERR("read fifo");
225 }
226 if (count == 0) {
227 LOG_DEBUG("All writers closed; reopening FIFO");
228 break;
229 }
230
231 len += (size_t)count;
232 buf[len] = '\0';
233
234 char *start = buf;
235 char *nl;
236 while ((nl = strchr(start, '\n')) != NULL) {
237 *nl = '\0';
238 if (handle_command(start, stepper)) {
239 done = 1;
240 break;
241 }
242 start = nl + 1;
243 }
244 if (done)
245 break;
246
247 len = strlen(start);
248 memmove(buf, start, len + 1);
249 if (len == sizeof(buf) - 1) {
250 LOG_WARN("Command buffer full without newline; dropping");
251 len = 0;
252 }
253 }
254 close(fifo);
255 }
256}

264int main(int argc, char **argv)
265{
266 stepper_t stepper;
267 int homed;
268
269 if (sprint1_logging_init("servo_calib", NULL, 0) != 0)
270 return EXIT_FAILURE;
271
272 if (argc != 1)
273 LOG_WARN("Invalid usage. Additional arguments are not supported. Being ignored.");
274
275 memset(&stepper, 0, sizeof(stepper));
276 stepper.fd = -1;
277 stepper.zero = 400;
278
279 LOG_INFO("Opening GPIO stepper driver (WiringPi)...");
280 if (!stepperOpen(NULL, &stepper)) {
281 LOG_FATAL("Failed to open stepper");
282 logger_flush();
283 return EXIT_FAILURE;
284 }
285
286 ms_sleep(100);
287 LOG_INFO("Homing (drive toward endstop)...");
288 homed = stepperInit(&stepper);
289 if (homed < 0) {
290 LOG_FATAL("Homing failed");
291 logger_flush();
292 stepperClose(&stepper);
293 return EXIT_FAILURE;
294 }
295 LOG_INFO("Homed position: %d (angle %.2f deg)", homed,
296 stepper_prismAngle(&stepper));
297 loadParams(PRM_FILE, &stepper);
298 stepper_setBrightness(&stepper, 255);
299 stepperSend(stepper.prep, &stepper);
300 LOG_INFO("Parked at prep=%d", stepper.prep);
301
302 serve_fifo(CMD_PIPE, &stepper);
303
304 stepperClose(&stepper);
305 LOG_INFO("Exiting");
306 logger_flush();
307 logger_exitFileLogger();
308 return 0;
309}
