153static void queue_goto(motor_t *stepper, int pos)
154{
155 int diff;
156
157 if (!stepper->init) {
158 stepper->error = 1;
159 return;
160 }
161
162 diff = pos - stepper->pos;
163 if (diff > 0) {
164 stepper->dir = 1;
165 stepper->steps = diff;
166 } else if (diff < 0) {
167 stepper->dir = 0;
168 stepper->steps = -diff;
169 } else {
170 stepper->steps = 0;
171 }
172 stepper->error = 0;
173}
179static void *motor_thread(void *arg)
180{
181 (void)arg;
182
183 while (g_thread_run) {
184 int do_step = 0;
185 int dir = 0;
186
187 pthread_mutex_lock(&g_mutex);
188 if (g_motor != NULL && g_motor->ena && g_motor->steps > 0) {
189 do_step = 1;
190 dir = g_motor->dir;
191 }
192 pthread_mutex_unlock(&g_mutex);
193
194 if (do_step) {
195 pulse_gpio(dir);
196
197 pthread_mutex_lock(&g_mutex);
198 if (g_motor != NULL) {
199 account_step(g_motor, dir);
200 if (g_motor->steps > 0) {
201 g_motor->steps--;
202 }
203 }
204 pthread_mutex_unlock(&g_mutex);
205 } else {
206 usleep(1000);
207 }
208 }
209
210 return NULL;
211}

227static int a4988_open_session(const char *portName, motor_t *m)
228{
229 (void)portName;
230
231 if (g_motor != NULL) {
232 LOG_ERROR("a4988_open: already open");
233 return -1;
234 }
235
236 if (wiringPiSetupPinType(WPI_PIN_BCM) < 0) {
237 LOG_ERROR("wiringPiSetupPinType failed (run with sudo)");
238 return -1;
239 }
240 LOG_INFO("WiringPi BCM pin mode ready");
241
242 claim_output(PIN_STEP, LOW);
243 claim_output(PIN_DIR, LOW);
244 claim_output(PIN_ENABLE, HIGH);
245 claim_output(PIN_MS1, LOW);
246 claim_output(PIN_MS2, LOW);
247 claim_output(PIN_MS3, LOW);
248 claim_input_pullup(PIN_ENDSTOP);
249 if (PIN_LID >= 0) {
250 claim_input_pullup(PIN_LID);
251 }
252
253 set_microstep_16();
254
255 m->fd = 1;
256 m->zero = CMD_POS_ZERO;
257 g_motor = m;
258
259 g_thread_run = 1;
260 if (pthread_create(&g_thread, NULL, motor_thread, NULL) != 0) {
261 ERR("pthread_create");
262 }
263
264 pthread_mutex_lock(&g_mutex);
265 disable_motor(m);
266 update_lid(m);
267 pthread_mutex_unlock(&g_mutex);
268
269 LOG_INFO("Stepper opened (motor thread running)");
270 return 0;
271}

293static int a4988_send(motor_t *m, int cmd)
294{
295 pthread_mutex_lock(&g_mutex);
296
297 if (cmd >= CMD_ANGLE_MIN && cmd <= CMD_ANGLE_MAX) {
298 queue_goto(m, cmd);
299 } else if (cmd == CMD_DISABLE) {
300 m->was_disabled = 1;
301 disable_motor(m);
302 } else if (cmd == CMD_ENABLE) {
303 if (m->was_disabled == 1) {
304 m->init = 1;
305 }
306 enable_motor(m);
307 } else if (cmd == CMD_RESET) {
308 /* Unreachable while CMD_RESET == CMD_DISABLE (both 1003). */
309 m->steps = 0;
310 m->error = 0;
311 } else if (cmd == CMD_NEXT) {
312 if (m->ena) {
313 pulse_step(m, 1);
314 m->dir = 1;
315 }
316 } else if (cmd == CMD_PREV) {
317 if (m->ena) {
318 pulse_step(m, 0);
319 m->dir = 0;
320 }
321 } else if (cmd == CMD_UPDATE_LID) {
322 update_lid(m);
323 } else {
324 LOG_WARN("a4988_send: unknown command %d", cmd);
325 }
326
327 pthread_mutex_unlock(&g_mutex);
328 return 1;
329}

348static int a4988_init(motor_t *m)
349{
350 int pos = -1;
351 int attempts = 0;
352
353 LOG_INFO("Homing: enable motor and seek endstop");
354 a4988_send(m, CMD_ENABLE);
355 ms_sleep(100);
356
357 while (pos < 0 && attempts < 2000) {
358 a4988_send(m, CMD_NEXT);
359 pos = a4988_read(m);
360 attempts++;
361 if ((attempts % 50) == 0) {
362 LOG_DEBUG("Homing... steps=%d endstop=%d", attempts,
363 digitalRead(PIN_ENDSTOP));
364 }
365 }
366 a4988_send(m, CMD_NEXT);
367 if (pos < 0) {
368 LOG_ERROR("Homing failed after %d steps (endstop never HIGH on GPIO %d)",
369 attempts, PIN_ENDSTOP);
370 return -1;
371 }
372
373 pthread_mutex_lock(&g_mutex);
374 m->zero = pos;
375 pthread_mutex_unlock(&g_mutex);
376
377 LOG_INFO("Endstop hit after %d steps, position=%d", attempts, pos);
378 return pos;
379}
