59static void draw_overlay(const cv::Mat &gray, const rois_t &rois, const char *path)
60{
61 cv::Mat vis;
62 cv::cvtColor(gray, vis, cv::COLOR_GRAY2BGR);
63 for (int i = 0; i < rois.n; i++) {
64 const roi_t *r = &rois.item[i];
65 cv::ellipse(vis, cv::Point((int)std::lround(r->cx), (int)std::lround(r->cy)),
66 cv::Size(std::max(1, (int)std::lround(r->rx)),
67 std::max(1, (int)std::lround(r->ry))),
68 r->angle, 0, 360, cv::Scalar(0, 255, 0), 2);
69 cv::putText(vis, r->name,
70 cv::Point((int)std::lround(r->cx - r->rx),
71 std::max(12, (int)std::lround(r->cy - r->ry) - 4)),
72 cv::FONT_HERSHEY_SIMPLEX, 0.6, cv::Scalar(0, 255, 255), 2);
73 }
74 cv::imwrite(path, vis);
75 LOG_INFO("wrote %s", path);
76}

81static int detect_at_pos(cam_t *cam, int pos, int nwant, rois_t *out,
82 const char *overlay_path, const char *frame_path)
83{
84 if (fifo_move_and_wait(pos) != 0) {
85 LOG_ERROR("move to %d failed", pos);
86 return -1;
87 }
88 usleep(50000); /* settle */
89
90 unsigned char *pix;
91 int w, h, pitch;
92 if (camera_api.grab(cam, &pix, &w, &h, &pitch) != 0)
93 return -1;
94
95 cv::Mat gray = frame_mat(pix, w, h, pitch);
96 if (frame_path)
97 cv::imwrite(frame_path, gray);
98
99 if (detect_boxes_cv(gray, nwant, out) != 0) {
100 LOG_ERROR("detect_boxes_cv failed at pos %d", pos);
101 return -1;
102 }
103 LOG_INFO("pos %d: found %d blob(s)", pos, out->n);
104 for (int i = 0; i < out->n; i++) {
105 LOG_INFO(" %s cx=%.1f cy=%.1f rx=%.1f ry=%.1f ang=%.1f", out->item[i].name,
106 out->item[i].cx, out->item[i].cy, out->item[i].rx, out->item[i].ry,
107 out->item[i].angle);
108 }
109 if (overlay_path)
110 draw_overlay(gray, *out, overlay_path);
111 return out->n > 0 ? 0 : -1;
112}

117static int write_interp_check_csv(cam_t *cam, const roi_interp_t *model, const char *csv_path)
118{
119 FILE *csv = fopen(csv_path, "w");
120 if (!csv) {
121 LOG_ERROR("%s: %s", csv_path, strerror(errno));
122 return -1;
123 }
124
125 fprintf(csv, "pos");
126 for (int i = 0; i < model->at0.n; i++)
127 fprintf(csv, ",%s_interp", model->at0.item[i].name);
128 for (int i = 0; i < model->at0.n; i++)
129 fprintf(csv, ",%s_fixed", model->at0.item[i].name);
130 for (int i = 0; i < model->at0.n; i++)
131 fprintf(csv, ",%s_cx,%s_cy", model->at0.item[i].name, model->at0.item[i].name);
132 fprintf(csv, "\n");
133
134 int lo = model->pos0 < model->pos1 ? model->pos0 : model->pos1;
135 int hi = model->pos0 < model->pos1 ? model->pos1 : model->pos0;
136
137 for (int pos = lo; pos <= hi; pos++) {
138 if (fifo_move_and_wait(pos) != 0) {
139 LOG_ERROR("move to %d failed during check", pos);
140 fclose(csv);
141 return -1;
142 }
143 usleep(20000);
144
145 unsigned char *pix;
146 int w, h, pitch;
147 if (camera_api.grab(cam, &pix, &w, &h, &pitch) != 0) {
148 fclose(csv);
149 return -1;
150 }
151
152 rois_t interp;
153 if (roi_interp_at(model, pos, &interp) != 0) {
154 fclose(csv);
155 return -1;
156 }
157 roi_clip(&interp, w, h);
158
159 rois_t fixed = model->at0;
160 roi_clip(&fixed, w, h);
161
162 double m_interp[ROI_MAX], m_fixed[ROI_MAX];
163 roi_means(pix, w, h, pitch, &interp, m_interp);
164 roi_means(pix, w, h, pitch, &fixed, m_fixed);
165
166 fprintf(csv, "%d", pos);
167 for (int i = 0; i < interp.n; i++)
168 fprintf(csv, ",%.4f", m_interp[i]);
169 for (int i = 0; i < fixed.n; i++)
170 fprintf(csv, ",%.4f", m_fixed[i]);
171 for (int i = 0; i < interp.n; i++)
172 fprintf(csv, ",%.3f,%.3f", interp.item[i].cx, interp.item[i].cy);
173 fprintf(csv, "\n");
174 fflush(csv);
175
176 LOG_INFO("check pos %d %s_interp=%.1f %s_fixed=%.1f", pos,
177 interp.item[0].name, m_interp[0], fixed.item[0].name, m_fixed[0]);
178 }
179
180 fclose(csv);
181 LOG_INFO("wrote intensity check CSV %s", csv_path);
182 return 0;
183}

185int main(int argc, char **argv)
186{
187 double exposure = argc > 1 ? strtod(argv[1], NULL) : 0.1;
188 int pos0 = argc > 2 ? atoi(argv[2]) : 400;
189 int pos1 = argc > 3 ? atoi(argv[3]) : 420;
190 int nwant = argc > 4 ? atoi(argv[4]) : 3;
191 const char *interp_path = argc > 5 ? argv[5] : ROI_INTERP_PATH;
192 const char *csv_path = argc > 6 ? argv[6] : "./tmp/interp_intensity.csv";
193
194 if (sprint1_logging_init("calibrate2", NULL, 0) != 0)
195 return 1;
196
197 if (pos0 == pos1) {
198 LOG_ERROR("pos0 and pos1 must differ");
199 return 1;
200 }
201
202 illum_on();
203
204 if (fifo_open() != 0)
205 return 1;
206
207 cam_t *cam = NULL;
208 if (camera_api.open(&cam) != 0) {
209 fifo_close();
210 return 1;
211 }
212 double actual = 0.0;
213 if (camera_api.set_exposure_ms(cam, exposure, &actual) != 0) {
214 camera_api.close(cam);
215 fifo_close();
216 return 1;
217 }
218 LOG_INFO("camera %s exposure %.3f ms; calibrate2 at %d and %d",
219 camera_api.sensor(cam), actual, pos0, pos1);
220
221 roi_interp_t model;
222 memset(&model, 0, sizeof(model));
223 model.pos0 = pos0;
224 model.pos1 = pos1;
225
226 char overlay0[128], overlay1[128], frame0[128], frame1[128];
227 snprintf(overlay0, sizeof(overlay0), "./tmp/detected_%d.png", pos0);
228 snprintf(overlay1, sizeof(overlay1), "./tmp/detected_%d.png", pos1);
229 snprintf(frame0, sizeof(frame0), "./tmp/calib_frame_%d.png", pos0);
230 snprintf(frame1, sizeof(frame1), "./tmp/calib_frame_%d.png", pos1);
231
232 if (detect_at_pos(cam, pos0, nwant, &model.at0, overlay0, frame0) != 0) {
233 camera_api.close(cam);
234 fifo_close();
235 return 1;
236 }
237 if (detect_at_pos(cam, pos1, nwant, &model.at1, overlay1, frame1) != 0) {
238 camera_api.close(cam);
239 fifo_close();
240 return 1;
241 }
242 if (model.at0.n != model.at1.n) {
243 LOG_ERROR("blob count mismatch: %d at %d vs %d at %d", model.at0.n, pos0,
244 model.at1.n, pos1);
245 camera_api.close(cam);
246 fifo_close();
247 return 1;
248 }
249
250 /* Log per-blob centre delta (should be small / nearly linear over 20 steps). */
251 for (int i = 0; i < model.at0.n; i++) {
252 LOG_INFO("%s delta: dcx=%.2f dcy=%.2f (%.2f px/step)", model.at0.item[i].name,
253 model.at1.item[i].cx - model.at0.item[i].cx,
254 model.at1.item[i].cy - model.at0.item[i].cy,
255 hypot(model.at1.item[i].cx - model.at0.item[i].cx,
256 model.at1.item[i].cy - model.at0.item[i].cy) /
257 (double)(pos1 - pos0));
258 }
259
260 if (roi_interp_save(interp_path, &model) != 0) {
261 camera_api.close(cam);
262 fifo_close();
263 return 1;
264 }
265 /* Also write a mid-point rois.json snapshot for manual inspection (not used by sweep yet). */
266 {
267 rois_t mid;
268 int midpos = (pos0 + pos1) / 2;
269 if (roi_interp_at(&model, midpos, &mid) == 0)
270 roi_save_json(ROI_PATH, &mid);
271 }
272
273 LOG_INFO("Sampling intensities pos %d..%d with interpolated ROIs (CSV for plot)",
274 pos0 < pos1 ? pos0 : pos1, pos0 < pos1 ? pos1 : pos0);
275 int rc = write_interp_check_csv(cam, &model, csv_path);
276
277 camera_api.close(cam);
278 fifo_close();
279 logger_flush();
280 logger_exitFileLogger();
281
282 if (rc == 0) {
283 fprintf(stderr,
284 "\nNext: plot intensity vs position to check for a flat slope:\n"
285 " python3 plot_interp_check.py %s ./tmp/interp_intensity.png\n\n",
286 csv_path);
287 }
288 return rc == 0 ? 0 : 1;
289}
