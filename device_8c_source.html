23{

25 char buffer[32];
26 int count;
27 int n;
28 int start, stop, prep;
29
30 if (fd == -1) {
31 LOG_ERROR("load_params: open(%s) failed: %s", filename, strerror(errno));
32 ERR("open");
33 }
34
35 memset(buffer, 0, sizeof(buffer));
36 count = read(fd, buffer, sizeof(buffer) - 1);
37 close(fd);
38
39 if (count <= 0) {
40 LOG_FATAL("load_params: empty or unreadable file %s", filename);
41 exit(EXIT_FAILURE);
42 }
43
44 buffer[count] = '\0';
45 n = sscanf(buffer, "%d %d %d", &start, &stop, &prep);
46 if (n != 3) {
47 LOG_FATAL("load_params: expected 3 integers, got %d", n);
48 exit(EXIT_FAILURE);
49 }
50
51 if (start < 0 || stop < 0 || prep < 0 ||
52 start > 999 || stop > 999 || prep > 999) {
53 LOG_FATAL("load_params: unsupported values start=%d stop=%d prep=%d",
54 start, stop, prep);
55 exit(EXIT_FAILURE);
56 }
57 motor_api.set_scan_params(motor, start, stop, prep);
58 LOG_INFO("Params: start=%d stop=%d prep=%d", start, stop, prep);
59}

82device_t *device_open(void)
83{
84 device_t *device = calloc(1, sizeof(*device));
85 if (!device)
86 return NULL;
87
88 if (sprint1_logging_init("driver", NULL, 0) != 0) {
89 free(device);
90 return NULL;
91 }
92
93 LOG_INFO("Opening motor driver...");
94 if (motor_api.open(&device->motor, NULL) != 0) {
95 LOG_FATAL("Failed to open motor");
96 device_close(device);
97 return NULL;
98 }
99
100 LOG_INFO("Opening illumination on pin %d...", PIN_LASER);
101 if (illumination_api.open(&device->illumination, PIN_LASER) != 0) {
102 LOG_FATAL("Failed to open illumination");
103 device_close(device);
104 return NULL;
105 }
106
107 ms_sleep(100);
108
109 LOG_INFO("Homing (drive toward endstop)...");
110 {
111 int homed = motor_api.init(device->motor);
112 if (homed < 0) {
113 LOG_FATAL("Homing failed");
114 device_close(device);
115 return NULL;
116 }
117 LOG_INFO("Homed position: %d (angle %.2f deg)", homed,
118 motor_api.prism_angle(device->motor));
119 }
120
121 load_params(PRM_FILE, device->motor);
122 illumination_api.set_brightness(device->illumination, 255);
123 illumination_api.on(device->illumination);
124
125 {
126 int prep = motor_api.get_prep(device->motor);
127 motor_api.send(device->motor, prep);
128 LOG_INFO("Parked at prep=%d", prep);
129 }
130
131 return device;
132}
