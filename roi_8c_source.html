35static int json_key_double(const char *obj, const char *end, const char *key, double *out)
36{
37 char pat[64];
38 snprintf(pat, sizeof(pat), "\"%s\"", key);
39 const char *p = obj;
40 while (p && p < end) {
41 p = strstr(p, pat);
42 if (!p || p >= end)
43 return -1;
44 p += strlen(pat);
45 p = skip_ws(p);
46 if (*p != ':')
47 continue;
48 p = skip_ws(p + 1);
49 char *next = NULL;
50 *out = strtod(p, &next);
51 if (next == p)
52 return -1;
53 return 0;
54 }
55 return -1;
56}

87static int point_in_ellipse(int x, int y, const roi_t *r)
88{
89 double dx = (x + 0.5) - r->cx;
90 double dy = (y + 0.5) - r->cy;
91 double ang = r->angle * (M_PI / 180.0);
92 double c = cos(ang);
93 double s = sin(ang);
94 double xr = dx * c + dy * s;
95 double yr = -dx * s + dy * c;
96 if (r->rx < 1e-6 || r->ry < 1e-6)
97 return 0;
98 return (xr * xr) / (r->rx * r->rx) + (yr * yr) / (r->ry * r->ry) <= 1.0;
99}

101void roi_sync_aabb(roi_t *r)
102{
103 double ang = r->angle * (M_PI / 180.0);
104 double c = cos(ang);
105 double s = sin(ang);
106 double ux = r->rx * c;
107 double uy = r->rx * s;
108 double vx = r->ry * -s;
109 double vy = r->ry * c;
110 double half_w = sqrt(ux * ux + vx * vx);
111 double half_h = sqrt(uy * uy + vy * vy);
112 int x0 = (int)floor(r->cx - half_w);
113 int y0 = (int)floor(r->cy - half_h);
114 int x1 = (int)ceil(r->cx + half_w);
115 int y1 = (int)ceil(r->cy + half_h);
116 r->x = x0;
117 r->y = y0;
118 r->w = x1 - x0;
119 r->h = y1 - y0;
120 if (r->w < 0)
121 r->w = 0;
122 if (r->h < 0)
123 r->h = 0;
124 r->pixels = (int)lround(M_PI * r->rx * r->ry);
125}

127void roi_inscribe(int bx, int by, int bw, int bh, roi_t *out)
128{
129 /*
130 * Ellipse filling (bx,by,bw,bh) has semi-axes bw/2, bh/2. The largest
131 * axis-aligned rectangle inside that ellipse scales the AABB by 1/sqrt(2)
132 * about its centre (AABB corners lie outside a disc).
133 */
134 double cx = bx + bw * 0.5;
135 double cy = by + bh * 0.5;
136 int w = (int)(bw / M_SQRT2 + 0.5);
137 int h = (int)(bh / M_SQRT2 + 0.5);
138 if (w < 1)
139 w = 1;
140 if (h < 1)
141 h = 1;
142 out->cx = cx;
143 out->cy = cy;
144 out->rx = w * 0.5;
145 out->ry = h * 0.5;
146 out->angle = 0.0;
147 out->w = w;
148 out->h = h;
149 out->x = (int)(cx - w * 0.5);
150 out->y = (int)(cy - h * 0.5);
151 out->pixels = w * h;
152}

154int roi_parse_json(const char *buf, rois_t *out)
155{
156 memset(out, 0, sizeof(*out));
157 if (!buf)
158 return -1;
159 const char *p = buf;
160 while (out->n < ROI_MAX) {
161 const char *start = strchr(p, '{');
162 if (!start)
163 break;
164 const char *stop = strchr(start, '}');
165 if (!stop)
166 break;
167 roi_t *r = &out->item[out->n];
168 memset(r, 0, sizeof(*r));
169 double cx, cy, rx, ry, angle = 0.0;
170 double x = 0, y = 0, w = 0, h = 0;
171 int got_ellipse = json_key_double(start, stop, "cx", &cx) == 0 &&
172 json_key_double(start, stop, "cy", &cy) == 0 &&
173 json_key_double(start, stop, "rx", &rx) == 0 &&
174 json_key_double(start, stop, "ry", &ry) == 0;
175 if (got_ellipse) {
176 (void)json_key_double(start, stop, "angle", &angle);
177 r->cx = cx;
178 r->cy = cy;
179 r->rx = rx;
180 r->ry = ry;
181 r->angle = angle;
182 roi_sync_aabb(r);
183 } else if (json_key_double(start, stop, "x", &x) == 0 &&
184 json_key_double(start, stop, "y", &y) == 0 &&
185 json_key_double(start, stop, "w", &w) == 0 &&
186 json_key_double(start, stop, "h", &h) == 0) {
187 /* Legacy box JSON: treat the rectangle as an axis-aligned ellipse. */
188 r->cx = x + w * 0.5;
189 r->cy = y + h * 0.5;
190 r->rx = w * 0.5;
191 r->ry = h * 0.5;
192 r->angle = 0.0;
193 roi_sync_aabb(r);
194 } else {
195 p = stop + 1;
196 continue;
197 }
198 if (json_key_string(start, stop, "name", r->name, sizeof(r->name)) != 0)
199 snprintf(r->name, sizeof(r->name), "ring%d", out->n + 1);
200 out->n++;
201 p = stop + 1;
202 }
203 return out->n > 0 ? 0 : -1;
204}

206int roi_load_json(const char *path, rois_t *out)
207{
208 FILE *f = fopen(path, "rb");
209 if (!f) {
210 LOG_ERROR("roi_load_json: open(%s): %s", path, strerror(errno));
211 return -1;
212 }
213 if (fseek(f, 0, SEEK_END) != 0) {
214 LOG_ERROR("roi_load_json: fseek(%s): %s", path, strerror(errno));
215 fclose(f);
216 return -1;
217 }
218 long sz = ftell(f);
219 if (sz < 2) {
220 fclose(f);
221 LOG_ERROR("%s: empty", path);
222 return -1;
223 }
224 rewind(f);
225 char *buf = malloc((size_t)sz + 1);
226 if (!buf) {
227 fclose(f);
228 LOG_ERROR("roi_load_json: out of memory");
229 return -1;
230 }
231 if (fread(buf, 1, (size_t)sz, f) != (size_t)sz) {
232 LOG_ERROR("roi_load_json: fread(%s): %s", path, strerror(errno));
233 free(buf);
234 fclose(f);
235 return -1;
236 }
237 fclose(f);
238 buf[sz] = '\0';
239
240 int rc = roi_parse_json(buf, out);
241 free(buf);
242 if (rc != 0) {
243 LOG_ERROR("%s: no ROIs found", path);
244 return -1;
245 }
246 LOG_INFO("Loaded %d ROI(s) from %s", out->n, path);
247 return 0;
248}

277void roi_clip(rois_t *rois, int width, int height)
278{
279 for (int i = 0; i < rois->n; i++) {
280 roi_t *r = &rois->item[i];
281 roi_sync_aabb(r);
282 if (r->x < 0) {
283 r->w += r->x;
284 r->x = 0;
285 }
286 if (r->y < 0) {
287 r->h += r->y;
288 r->y = 0;
289 }
290 if (r->x + r->w > width)
291 r->w = width - r->x;
292 if (r->y + r->h > height)
293 r->h = height - r->y;
294 if (r->w < 0)
295 r->w = 0;
296 if (r->h < 0)
297 r->h = 0;
298
299 int n = 0;
300 for (int y = r->y; y < r->y + r->h; y++) {
301 for (int x = r->x; x < r->x + r->w; x++) {
302 if (point_in_ellipse(x, y, r))
303 n++;
304 }
305 }
306 r->pixels = n;
307 }

313static int accum_blob(const unsigned char *pixels, int width, int height, int pitch,
314 const roi_t *r, unsigned long *sum, int *n, int sat_level,
315 unsigned long *nsat)
316{
317 *sum = 0;
318 *n = 0;
319 if (nsat)
320 *nsat = 0;
321 if (r->w < 1 || r->h < 1 || r->rx < 1e-6 || r->ry < 1e-6)
322 return 0;
323 for (int y = r->y; y < r->y + r->h && y < height; y++) {
324 if (y < 0)
325 continue;
326 const unsigned char *row = pixels + (size_t)y * (size_t)pitch;
327 for (int x = r->x; x < r->x + r->w; x++) {
328 if (x < 0 || x >= width)
329 continue;
330 if (!point_in_ellipse(x, y, r))
331 continue;
332 unsigned char v = row[x];
333 *sum += v;
334 (*n)++;
335 if (nsat && v >= (unsigned char)sat_level)
336 (*nsat)++;
337 }
338 }
339 return 0;
340}

367void roi_draw(unsigned char *pixels, int width, int height, int pitch,
368 const rois_t *rois)
369{
370 for (int i = 0; i < rois->n; i++) {
371 const roi_t *r = &rois->item[i];
372 /* Outline: pixels near the ellipse boundary (1.0 ± epsilon). */
373 for (int y = r->y; y < r->y + r->h && y < height; y++) {
374 if (y < 0)
375 continue;
376 for (int x = r->x; x < r->x + r->w; x++) {
377 if (x < 0 || x >= width)
378 continue;
379 double dx = (x + 0.5) - r->cx;
380 double dy = (y + 0.5) - r->cy;
381 double ang = r->angle * (M_PI / 180.0);
382 double c = cos(ang);
383 double s = sin(ang);
384 double xr = dx * c + dy * s;
385 double yr = -dx * s + dy * c;
386 if (r->rx < 1e-6 || r->ry < 1e-6)
387 continue;
388 double v = (xr * xr) / (r->rx * r->rx) + (yr * yr) / (r->ry * r->ry);
389 if (v >= 0.92 && v <= 1.08)
390 pixels[(size_t)y * (size_t)pitch + (size_t)x] = 255;
391 }
392 }
393 }
