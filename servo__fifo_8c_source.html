22int servo_fifo_open(void)
23{
24 if (g_cmd_fd >= 0)
25 return 0;
26 signal(SIGPIPE, SIG_IGN);
27 for (int attempt = 0; attempt < 50; attempt++) {
28 g_cmd_fd = open(CMD_PIPE, O_WRONLY);
29 if (g_cmd_fd >= 0) {
30 LOG_INFO("Opened command FIFO %s", CMD_PIPE);
31 break;
32 }
33 if (errno != EINTR && errno != ENXIO) {
34 LOG_ERROR("%s: %s", CMD_PIPE, strerror(errno));
35 return -1;
36 }
37 usleep(50000);
38 }
39 if (g_cmd_fd < 0) {
40 LOG_ERROR("%s: no reader (is servo_calib running?)", CMD_PIPE);
41 return -1;
42 }
43 if (g_pos_fd < 0) {
44 for (int attempt = 0; attempt < 50; attempt++) {
45 g_pos_fd = open(POS_PIPE, O_RDONLY | O_NONBLOCK);
46 if (g_pos_fd >= 0) break;
47 if (errno != EINTR && errno != ENXIO && errno != EBADF) {
48 LOG_ERROR("%s: %s", POS_PIPE, strerror(errno));
49 close(g_cmd_fd);
50 g_cmd_fd = -1;
51 return -1;
52 }
53 usleep(50000);
54 }
55 if (g_pos_fd < 0) {
56 LOG_ERROR("%s: no reader", POS_PIPE);
57 close(g_cmd_fd);
58 g_cmd_fd = -1;
59 return -1;
60 }
61 }
62 return 0;
63}

65int servo_fifo_send(long cmd)
66{
67 char buf[32];
68 int n = snprintf(buf, sizeof(buf), "%ld\n", cmd);
69
70 for (int attempt = 0; attempt < 20; attempt++) {
71 if (servo_fifo_open() != 0)
72 return -1;
73 ssize_t w = write(g_cmd_fd, buf, (size_t)n);
74 if (w == (ssize_t)n) {
75 LOG_DEBUG("FIFO send %ld", cmd);
76 return 0;
77 }
78 if (w < 0 && (errno == EPIPE || errno == EINTR || errno == EAGAIN)) {
79 close(g_cmd_fd);
80 g_cmd_fd = -1;
81 usleep(20000);
82 continue;
83 }
84 LOG_ERROR("write cmd: %s", strerror(errno));
85 return -1;
86 }
87 LOG_ERROR("write cmd: giving up after retries (cmd=%ld)", cmd);
88 return -1;

91static int position_once(int timeout_ms)
92{
93 if (g_pos_fd < 0) {
94 if (servo_fifo_open() != 0)
95 return -1;
96 }
97 char junk[256];
98 for (;;) {
99 ssize_t n = read(g_pos_fd, junk, sizeof(junk));
100 if (n <= 0) break;
101 }
102 if (servo_fifo_send(SERVO_CMD_GETPOS) != 0) {
103 return -1;
104 }
105
106 char buf[64]={};
107 size_t len = 0;
108 while (memchr(buf, '\n', len) == NULL) {
109 struct pollfd pfd = {.fd = g_pos_fd, .events = POLLIN};
110 int pr = poll(&pfd, 1, timeout_ms);
111 if (pr <= 0) {
112 return -1;
113 }
114 ssize_t n = read(g_pos_fd, buf + len, sizeof(buf) - 1 - len);
115 if (n < 0) {
116 if (errno == EAGAIN || errno == EWOULDBLOCK) {
117 usleep(5000);
118 continue;
119 }
120 return -1;
121 }
122 if (n == 0) {
123 usleep(5000);
124 continue;
125 }
126 len += (size_t)n;
127 if (len >= sizeof(buf) - 1) {
128 return -1;
129 }
130 }
131 buf[len] = '\0';
