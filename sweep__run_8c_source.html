46static int load_prm(const char *path, int *start, int *stop, int *prep)
47{
48 FILE *f = fopen(path, "r");
49 if (!f) {
50 LOG_ERROR("%s: %s", path, strerror(errno));
51 return -1;
52 }
53 int n = fscanf(f, "%d %d %d", start, stop, prep);
54 fclose(f);
55 if (n != 3) {
56 LOG_ERROR("%s: expected 3 integers", path);
57 return -1;
58 }
59 return 0;
60}

80static int do_sweep(void *userdata)
81{
82 sweep_ctx_t *ctx = userdata;
83 FILE *csv;
84 unsigned char *pix;
85 int w, h, pitch;
86 double means[ROI_MAX];
87 double sat[ROI_MAX];
88 int total;
89 int done_steps = 0;
90 char d1[8], d2[8];
91
92 disease_label(ctx->assign->blob1, d1, sizeof(d1));
93 disease_label(ctx->assign->blob2, d2, sizeof(d2));
94
95 sweep_tui_set_running("Starting scan…", "Moving to the first position.");
96 sweep_tui_set_progress(0, NULL);
97
98 csv = fopen(ctx->csv_path, "w");
99 if (!csv) {
100 LOG_ERROR("%s: %s", ctx->csv_path, strerror(errno));
101 sweep_tui_set_error("Could not create the results file.", ctx->csv_path);
102 return -1;
103 }
104 fprintf(csv, "Angle_deg");
105 for (int i = 0; i < ctx->rois->n; i++)
106 fprintf(csv, ",%s", ctx->rois->item[i].name);
107 fprintf(csv, ",blob1_disease,blob2_disease\n");
108 fflush(csv);
109
110 if (servo_fifo_move_and_wait(ctx->start) != 0) {
111 LOG_ERROR("move_and_wait(%d) failed", ctx->start);
112 fclose(csv);
113 sweep_tui_set_error("Could not move into place.",
114 "Check that servo_calib is running.");
115 return -1;
116 }
117
118 if (cam_grab(ctx->cam, &pix, &w, &h, &pitch) != 0) {
119 fclose(csv);
120 sweep_tui_set_error("Camera grab failed.", "Check the camera connection.");
121 return -1;
122 }
123 roi_clip(ctx->rois, w, h);
124
125 roi_means(pix, w, h, pitch, ctx->rois, means);
126 roi_sat_frac(pix, w, h, pitch, ctx->rois, 250, sat);
127 for (int i = 0; i < ctx->rois->n; i++) {
128 if (sat[i] > 0.05)
129 LOG_WARN("%s is %.0f%% saturated; lower exposure",
130 ctx->rois->item[i].name, sat[i] * 100.0);
131 }
132
133 total = ctx->stop - ctx->start + 1;
134 if (total < 1)
135 total = 1;
136
137 LOG_INFO("Sweep starting: %d steps (blob1=%s blob2=%s)", total, d1, d2);
138 for (int pos = ctx->start; pos <= ctx->stop; pos++) {
139 char detail[160];
140
141 if (servo_fifo_move_and_wait(pos) != 0) {
142 LOG_ERROR("move_and_wait(%d) failed", pos);
143 fclose(csv);
144 sweep_tui_set_error("Scan stopped — motor did not arrive.",
145 "Try Reset, then Start again.");
146 return -1;
147 }
148 if (cam_grab(ctx->cam, &pix, &w, &h, &pitch) != 0) {
149 fclose(csv);
150 sweep_tui_set_error("Camera grab failed during the scan.",
151 "Try Reset, then Start again.");
152 return -1;
153 }
154 roi_means(pix, w, h, pitch, ctx->rois, means);
155
156 double angle = pos * DEG_PER_STEP;
157 fprintf(csv, "%.4f", angle);
158 for (int i = 0; i < ctx->rois->n; i++)
159 fprintf(csv, ",%.4f", means[i]);
160 fprintf(csv, ",%s,%s\n", d1, d2);
161 fflush(csv);
162
163 done_steps++;
164 snprintf(detail, sizeof(detail), "Measuring… %d of %d (%.1f°)",
165 done_steps, total, angle);
166 sweep_tui_set_running("Scan in progress", detail);
167 sweep_tui_set_progress((done_steps * 100) / total, detail);
168
169 LOG_INFO("pos %d angle %.4f means[0]=%.1f", pos, angle,
170 ctx->rois->n > 0 ? means[0] : 0.0);
171 }
172
173 fclose(csv);
174 {
175 char detail[160];
176 snprintf(detail, sizeof(detail), "Saved to %s", ctx->csv_path);
177 sweep_tui_set_done("All done!", detail);
178 }
179 LOG_INFO("wrote %s", ctx->csv_path);
180 return 0;
181}

183int main(int argc, char **argv)
184{
185 const char *roi_path = argc > 1 ? argv[1] : ROI_PATH;
186 double exposure = argc > 2 ? strtod(argv[2], NULL) : 0.1;
187 const char *csv_path = argc > 3 ? argv[3] : "../sensor_data.csv";
188
189 cam_t cam;
190 rois_t rois;
191 disease_assign_t assign;
192 sweep_ctx_t ctx;
193 sweep_tui_hooks_t hooks;
194 double actual = 0.0;
195 int rc;
196
197 memset(&assign, 0, sizeof(assign));
198 memset(&ctx, 0, sizeof(ctx));
199 memset(&hooks, 0, sizeof(hooks));
200
201 if (sprint1_logging_init("sweep_run", NULL, 0) != 0)
202 return 1;
203
204 if (roi_load_json(roi_path, &rois) != 0)
205 return 1;
206
207 if (load_prm(PRM_FILE, &ctx.start, &ctx.stop, &ctx.prep) != 0)
208 return 1;
209 LOG_INFO("scan %d .. %d (prep %d) from %s", ctx.start, ctx.stop, ctx.prep,
210 PRM_FILE);
211
212 if (cam_open(&cam) != 0)
213 return 1;
214
215 if (cam_set_exposure_ms(&cam, exposure, &actual) != 0) {
216 cam_close(&cam);
217 return 1;
218 }
219
220 if (servo_fifo_open() != 0) {
221 cam_close(&cam);
222 return 1;
223 }
224
225 ctx.cam = &cam;
226 ctx.rois = &rois;
227 ctx.assign = &assign;
228 ctx.csv_path = csv_path;
229
230 hooks.on_start = do_sweep;
231 hooks.on_reset = do_reset;
232 hooks.userdata = &ctx;
233 hooks.scan_start = ctx.start;
234 hooks.scan_stop = ctx.stop;
235 hooks.scan_prep = ctx.prep;
236
237 rc = sweep_tui_run(&assign, &hooks);
238
239 servo_fifo_close();
240 cam_close(&cam);
241 LOG_INFO("TUI exit (servo daemon left running)");
242 logger_flush();
243 logger_exitFileLogger();
244 return rc;
245}
