44static int fifo_open_cmd(void)
45{
46 if (g_cmd_fd >= 0)
47 return 0;
48 /* Without this, EPIPE also delivers SIGPIPE and kills the process. */
49 signal(SIGPIPE, SIG_IGN);
50 for (int attempt = 0; attempt < 50; attempt++) {
51 g_cmd_fd = open(CMD_PIPE, O_WRONLY);
52 if (g_cmd_fd >= 0) {
53 LOG_INFO("Opened command FIFO %s", CMD_PIPE);
54 return 0;
55 }
56 if (errno != EINTR && errno != ENXIO) {
57 LOG_ERROR("%s: %s", CMD_PIPE, strerror(errno));
58 return -1;
59 }
60 usleep(50000);
61 }
62 LOG_ERROR("%s: no reader (is servo_calib running?)", CMD_PIPE);
63 return -1;
64}

71static int fifo_send(long cmd)
72{
73 char buf[32];
74 int n = snprintf(buf, sizeof(buf), "%ld\n", cmd);
75
76 for (int attempt = 0; attempt < 20; attempt++) {
77 if (fifo_open_cmd() != 0)
78 return -1;
79 ssize_t w = write(g_cmd_fd, buf, (size_t)n);
80 if (w == (ssize_t)n) {
81 LOG_DEBUG("FIFO send %ld", cmd);
82 return 0;
83 }
84 if (w < 0 && (errno == EPIPE || errno == EINTR || errno == EAGAIN)) {
85 close(g_cmd_fd);
86 g_cmd_fd = -1;
87 usleep(20000);
88 continue;
89 }
90 LOG_ERROR("write cmd: %s", strerror(errno));
91 return -1;
92 }
93 LOG_ERROR("write cmd: giving up after retries (cmd=%ld)", cmd);
94 return -1;
95}

116static int fifo_position_once(int timeout_ms)
117{
118 int fd = open(POS_PIPE, O_RDONLY | O_NONBLOCK);
119 if (fd < 0) {
120 if (errno != ENXIO)
121 LOG_ERROR("%s: %s", POS_PIPE, strerror(errno));
122 return -1;
123 }
124 if (fifo_send(CMD_GETPOS) != 0) {
125 close(fd);
126 return -1;
127 }
128
129 char buf[64]={};
130 size_t len = 0;
131 while (memchr(buf, '\n', len) == NULL) {
132 struct pollfd pfd = { .fd = fd, .events = POLLIN };
133 int pr = poll(&pfd, 1, timeout_ms);
134 if (pr <= 0) {
135 close(fd);
136 return -1;
137 }
138 ssize_t n = read(fd, buf + len, sizeof(buf) - 1 - len);
139 if (n < 0) {
140 if (errno == EAGAIN || errno == EWOULDBLOCK) {
141 usleep(5000);
142 continue;
143 }
144 close(fd);
145 return -1;
146 }
147 if (n == 0) {
148 usleep(5000);
149 continue;
150 }
151 len += (size_t)n;
152 if (len >= sizeof(buf) - 1) {
153 close(fd);
154 return -1;
155 }
156 }
157 close(fd);
158 buf[len] = '\0';
159 return atoi(buf);
160}

183static int move_and_wait(int pos)
184{
185 LOG_DEBUG("move_and_wait(%d)", pos);
186 if (fifo_send(pos) != 0)
187 return -1;
188 for (int spins = 0; spins < 6000; spins++) {
189 int cur = fifo_position();

191 return -1;

194 usleep(20000);
195 }
