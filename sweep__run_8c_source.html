132static int do_sweep(void)
133{
134 FILE *csv;
135 unsigned char *pix;
136 int w, h, pitch;
137 double means[ROI_MAX];
138 double sat[ROI_MAX];
139 int total;
140 int done_steps = 0;
141
142 set_status(ST_RUNNING, "Starting scan…", "Moving to the first position.");
143 g.progress = 0;
144 bump();
145
146 csv = fopen(g.csv_path, "w");
147 if (!csv) {
148 LOG_ERROR("%s: %s", g.csv_path, strerror(errno));
149 set_status(ST_ERROR, "Could not create the results file.", g.csv_path);
150 bump();
151 return -1;
152 }
153 fprintf(csv, "Angle_deg");
154 for (int i = 0; i < g.rois->n; i++)
155 fprintf(csv, ",%s", g.rois->item[i].name);
156 fprintf(csv, "\n");
157 fflush(csv);
158
159 if (servo_fifo_move_and_wait(g.start) != 0) {
160 LOG_ERROR("move_and_wait(%d) failed", g.start);
161 fclose(csv);
162 set_status(ST_ERROR, "Could not move into place.",
163 "Check that servo_calib is running.");
164 bump();
165 return -1;
166 }
167
168 if (cam_grab(g.cam, &pix, &w, &h, &pitch) != 0) {
169 fclose(csv);
170 set_status(ST_ERROR, "Camera grab failed.", "Check the camera connection.");
171 bump();
172 return -1;
173 }
174 roi_clip(g.rois, w, h);
175
176 roi_means(pix, w, h, pitch, g.rois, means);
177 roi_sat_frac(pix, w, h, pitch, g.rois, 250, sat);
178 for (int i = 0; i < g.rois->n; i++) {
179 if (sat[i] > 0.05)
180 LOG_WARN("%s is %.0f%% saturated; lower exposure", g.rois->item[i].name,
181 sat[i] * 100.0);
182 }
183
184 total = g.stop - g.start + 1;
185 if (total < 1)
186 total = 1;
187
188 LOG_INFO("Sweep starting: %d steps", total);
189 for (int pos = g.start; pos <= g.stop; pos++) {
190 char detail[160];
191
192 if (servo_fifo_move_and_wait(pos) != 0) {
193 LOG_ERROR("move_and_wait(%d) failed", pos);
194 fclose(csv);
195 set_status(ST_ERROR, "Scan stopped — motor did not arrive.",
196 "Try Reset, then Start again.");
197 bump();
198 return -1;
199 }
200 if (cam_grab(g.cam, &pix, &w, &h, &pitch) != 0) {
201 fclose(csv);
202 set_status(ST_ERROR, "Camera grab failed during the scan.",
203 "Try Reset, then Start again.");
204 bump();
205 return -1;
206 }
207 roi_means(pix, w, h, pitch, g.rois, means);
208
209 double angle = pos * DEG_PER_STEP;
210 fprintf(csv, "%.4f", angle);
211 for (int i = 0; i < g.rois->n; i++)
212 fprintf(csv, ",%.4f", means[i]);
213 fprintf(csv, "\n");
214 fflush(csv);
215
216 done_steps++;
217 g.progress = (done_steps * 100) / total;
218 snprintf(detail, sizeof(detail), "Measuring… %d of %d (%.1f°)",
219 done_steps, total, angle);
220 set_status(ST_RUNNING, "Scan in progress", detail);
221 bump();
222
223 LOG_INFO("pos %d angle %.4f means[0]=%.1f", pos, angle,
224 g.rois->n > 0 ? means[0] : 0.0);
225 }
226
227 fclose(csv);
228 g.progress = 100;
229 {
230 char detail[160];
231 snprintf(detail, sizeof(detail), "Saved to %s", g.csv_path);
232 set_status(ST_DONE, "All done!", detail);
233 }
234 bump();
235 LOG_INFO("wrote %s", g.csv_path);
236 return 0;
237}

503int main(int argc, char **argv)
504{
505 const char *roi_path = argc > 1 ? argv[1] : ROI_PATH;
506 double exposure = argc > 2 ? strtod(argv[2], NULL) : 0.1;
507 const char *csv_path = argc > 3 ? argv[3] : "../sensor_data.csv";
508
509 cam_t cam;
510 rois_t rois;
511 ui_t ui;
512 double actual = 0.0;
513 int last_cols = 0;
514 int last_rows = 0;
515
516 memset(&g, 0, sizeof(g));
517 g.csv_path = csv_path;
518 g.exposure = exposure;
519
520 if (sprint1_logging_init("sweep_run", NULL, 0) != 0)
521 return 1;
522
523 if (roi_load_json(roi_path, &rois) != 0)
524 return 1;
525 g.rois = &rois;
526
527 if (load_prm(PRM_FILE, &g.start, &g.stop, &g.prep) != 0)
528 return 1;
529 LOG_INFO("scan %d .. %d (prep %d) from %s", g.start, g.stop, g.prep, PRM_FILE);
530
531 if (cam_open(&cam) != 0)
532 return 1;
533 g.cam = &cam;
534
535 if (cam_set_exposure_ms(&cam, exposure, &actual) != 0) {
536 cam_close(&cam);
537 return 1;
538 }
539
540 if (servo_fifo_open() != 0) {
541 cam_close(&cam);
542 return 1;
543 }
544
545 ui_new(0, &ui);
546 ui.canscroll = 0;
547 g.ui = &ui;
548 g.panel_w = panel_width(&ui);
549 last_cols = ui.ws.ws_col;
550 last_rows = ui.ws.ws_row;
551
552 set_status(ST_READY, "Ready to scan.", "Press Start or S to let it move");
553 build_ui(&ui);
554 ui_draw(&ui);
555
556 {
557 char buf[64];
558 int n;
559 while (!g.quit && (n = read(STDIN_FILENO, buf, sizeof(buf))) > 0) {
560 ui_refresh_size(&ui);
561 if (ui.ws.ws_col != last_cols || ui.ws.ws_row != last_rows) {
562 last_cols = ui.ws.ws_col;
563 last_rows = ui.ws.ws_row;
564 relayout(&ui);
565 }
566 _ui_update(buf, n, &ui);
567 }
568 }
569
570 g.ui = NULL;
571 ui_free(&ui);
572
573 servo_fifo_close();
574 cam_close(&cam);
575 LOG_INFO("TUI exit (servo daemon left running)");
576 logger_flush();
577 logger_exitFileLogger();
578 return (g.state == ST_ERROR) ? 1 : 0;
579}
