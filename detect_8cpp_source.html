49static cv::Mat chip_mask(const cv::Mat &small, double frac, bool hull)
50{
51 cv::Mat bg;
52 cv::blur(small, bg, cv::Size(31, 31));
53
54 double p99;
55 {
56 std::vector<uchar> buf(bg.begin<uchar>(), bg.end<uchar>());
57 std::nth_element(buf.begin(), buf.begin() + (int)(0.99 * (buf.size() - 1)), buf.end());
58 p99 = buf[(int)(0.99 * (buf.size() - 1))];
59 }
60
61 cv::Mat mask = bg > (frac * p99);
62 mask.convertTo(mask, CV_8U, 255.0);
63
64 cv::Mat labels, stats, cents;
65 int n = cv::connectedComponentsWithStats(mask, labels, stats, cents, 8, CV_32S);
66 if (n <= 1)
67 return cv::Mat::ones(small.size(), CV_8U);
68
69 int best = 1, best_a = 0;
70 for (int i = 1; i < n; i++) {
71 int a = stats.at<int>(i, cv::CC_STAT_AREA);
72 if (a > best_a) {
73 best_a = a;
74 best = i;
75 }
76 }
77 mask = (labels == best);
78 mask.convertTo(mask, CV_8U, 255.0);
79
80 if (hull) {
81 std::vector<std::vector<cv::Point>> cnts;
82 cv::findContours(mask, cnts, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
83 mask = cv::Mat::zeros(small.size(), CV_8U);
84 if (!cnts.empty()) {
85 auto it = std::max_element(cnts.begin(), cnts.end(),
86 [](const auto &a, const auto &b) {
87 return cv::contourArea(a) < cv::contourArea(b);
88 });
89 std::vector<cv::Point> hull_pts;
90 cv::convexHull(*it, hull_pts);
91 cv::fillConvexPoly(mask, hull_pts, 255);
92 }
93 }
94
95 cv::Mat k = cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(9, 9));
96 cv::erode(mask, mask, k);
97 return mask;
98}

104static cv::Mat flatten(const cv::Mat &gray, double r_hi)
105{
106 int bk = ((int)(2.0 * r_hi) | 1);
107 if (bk < 51)
108 bk = 51;
109 if (bk > 151)
110 bk = 151; /* Cap kernel size — large blurs are slow on the Pi. */
111 cv::Mat bg;
112 cv::GaussianBlur(gray, bg, cv::Size(bk, bk), 0);
113 cv::Mat g32, b32, out;
114 gray.convertTo(g32, CV_32F);
115 bg.convertTo(b32, CV_32F);
116 cv::Mat n = g32 / (b32 + 1.f) * 128.f;
117 n.convertTo(out, CV_8U);
118 return out;
119}

135static EllipseCand refine_disc(const cv::Mat &sm, EllipseCand e)
136{
137 int h = sm.rows, w = sm.cols;
138 double r = std::max(e.rx, e.ry);
139 int x0 = std::max(0, (int)(e.cx - 1.8 * r));
140 int y0 = std::max(0, (int)(e.cy - 1.8 * r));
141 int x1 = std::min(w, (int)(e.cx + 1.8 * r));
142 int y1 = std::min(h, (int)(e.cy + 1.8 * r));
143 if (x1 <= x0 || y1 <= y0)
144 return e;
145
146 cv::Mat crop = sm(cv::Rect(x0, y0, x1 - x0, y1 - y0));
147 double lcx = e.cx - x0, lcy = e.cy - y0;
148
149 cv::Mat inner = cv::Mat::zeros(crop.size(), CV_8U);
150 cv::Mat out_big = cv::Mat::zeros(crop.size(), CV_8U);
151 cv::Mat out_small = cv::Mat::zeros(crop.size(), CV_8U);
152 draw_ellipse_mask(inner, lcx, lcy, e.rx, e.ry, e.angle, 0.7);
153 draw_ellipse_mask(out_big, lcx, lcy, e.rx, e.ry, e.angle, 1.6);
154 draw_ellipse_mask(out_small, lcx, lcy, e.rx, e.ry, e.angle, 1.2);
155 cv::Mat ring_out = out_big & ~out_small;
156
157 if (cv::countNonZero(inner) < 20 || cv::countNonZero(ring_out) < 20)
158 return e;
159
160 std::vector<uchar> iv, ov;
161 for (int y = 0; y < crop.rows; y++) {
162 const uchar *row = crop.ptr<uchar>(y);
163 const uchar *ii = inner.ptr<uchar>(y);
164 const uchar *oo = ring_out.ptr<uchar>(y);
165 for (int x = 0; x < crop.cols; x++) {
166 if (ii[x])
167 iv.push_back(row[x]);
168 if (oo[x])
169 ov.push_back(row[x]);
170 }
171 }
172 if (iv.size() < 20 || ov.size() < 20)
173 return e;
174
175 std::nth_element(iv.begin(), iv.begin() + (int)(0.75 * (iv.size() - 1)), iv.end());
176 double lo = iv[(int)(0.75 * (iv.size() - 1))];
177 std::nth_element(ov.begin(), ov.begin() + ov.size() / 2, ov.end());
178 double hi = ov[ov.size() / 2];
179 if (hi <= lo)
180 return e;
181 e.score = 1.0 - lo / hi;
182 double thr = 0.5 * (lo + hi);
183
184 cv::Mat limit = cv::Mat::zeros(crop.size(), CV_8U);
185 draw_ellipse_mask(limit, lcx, lcy, e.rx, e.ry, e.angle, 1.4);
186 cv::Mat blob = (crop < thr) & limit;
187
188 cv::Mat labels, stats, cents;
189 int nlab = cv::connectedComponentsWithStats(blob, labels, stats, cents, 8, CV_32S);
190 if (nlab <= 1)
191 return e;
192
193 int cy_i = (int)lcy, cx_i = (int)lcx;
194 int lab = 0;
195 if (cy_i >= 0 && cy_i < crop.rows && cx_i >= 0 && cx_i < crop.cols)
196 lab = labels.at<int>(cy_i, cx_i);
197 if (lab == 0) {
198 int best = 1, best_a = 0;
199 for (int i = 1; i < nlab; i++) {
200 int a = stats.at<int>(i, cv::CC_STAT_AREA);
201 if (a > best_a) {
202 best_a = a;
203 best = i;
204 }
205 }
206 lab = best;
207 }
208
209 cv::Mat comp = (labels == lab);
210 comp.convertTo(comp, CV_8U, 255.0);
211 std::vector<std::vector<cv::Point>> cnts;
212 cv::findContours(comp, cnts, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
213 if (cnts.empty())
214 return e;
215 auto c = *std::max_element(cnts.begin(), cnts.end(),
216 [](const auto &a, const auto &b) {
217 return cv::contourArea(a) < cv::contourArea(b);
218 });
219 if (c.size() < 5)
220 return e;
221
222 cv::RotatedRect rr = cv::fitEllipse(c);
223 EllipseCand out;
224 out.cx = rr.center.x + x0;
225 out.cy = rr.center.y + y0;
226 out.rx = rr.size.width / 2.0;
227 out.ry = rr.size.height / 2.0;
228 out.angle = rr.angle;
229 double area = cv::contourArea(c);
230 out.coverage = (rr.size.width * rr.size.height > 0)
231 ? area / (M_PI * rr.size.width * rr.size.height / 4.0)
232 : 0.0;
233 out.score = e.score;
234
235 double area_coarse = M_PI * e.rx * e.ry;
236 double area_new = M_PI * out.rx * out.ry;
237 double dist = std::hypot(out.cx - e.cx, out.cy - e.cy);
238 /* Accept modest centre drift; left-side discs often shift after refine. */
239 if (!(0.55 < area_new / area_coarse && area_new / area_coarse < 1.55) ||
240 dist > 0.4 * r)
241 return e;
242 return out;
243}

246static bool overlaps(const EllipseCand &a, const EllipseCand &b)
247{

262int detect_boxes_cv(const cv::Mat &gray, int nwant, rois_t *out)
263{
264 std::memset(out, 0, sizeof(*out));
265 if (gray.empty() || gray.type() != CV_8UC1) {
266 LOG_ERROR("detect_boxes_cv: empty or non-gray frame");
267 return -1;
268 }
269 if (nwant < 1)
270 nwant = 1;
271 if (nwant > ROI_MAX)
272 nwant = ROI_MAX;
273
274 LOG_INFO("detect_boxes_cv: %dx%d want=%d", gray.cols, gray.rows, nwant);
275
276 int h = gray.rows, w = gray.cols;
277 /* Discs are ~80–110 px radius on a 600×740 crop; crosses are much smaller. */
278 double r_lo = 0.09 * std::min(h, w);
279 double r_hi = 0.20 * std::min(h, w);
280 double area_lo = M_PI * r_lo * r_lo * 0.85; /* Reject cross-sized blobs. */
281 double area_hi = M_PI * r_hi * r_hi * 1.3;
282
283 cv::Mat flat = flatten(gray, r_hi);
284 int k = std::max(3, ((int)(0.18 * r_lo) | 1));
285 cv::Mat sm;
286 cv::blur(flat, sm, cv::Size(k, k));
287
288 cv::Mat small;
289 cv::resize(flat, small, cv::Size(), 0.25, 0.25, cv::INTER_AREA);
290 cv::Mat chip = chip_mask(small, 0.45, true);
291 cv::resize(chip, chip, gray.size(), 0, 0, cv::INTER_NEAREST);
292 /* Mild erode keeps the top-left disc inside the chip mask. */
293 int er = std::max(3, ((int)(0.22 * r_lo) | 1));
294 cv::erode(chip, chip, cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(er, er)));
295 if (cv::countNonZero(chip) == 0)
296 return 0;
297
298 /* Multi-level thresholds relative to flattened background (~128). */
299 const double fracs[] = {0.78, 0.82, 0.86, 0.88, 0.92};
300 std::vector<EllipseCand> found;
301 for (double frac : fracs) {
302 cv::Mat dark = (sm < frac * 128.0) & chip;
303 dark.convertTo(dark, CV_8U, 255.0);
304 int ko_s = std::max(3, ((int)(0.18 * r_lo) | 1));
305 /* Strong close fills streaky disc interiors (especially the bottom mark). */
306 int kc_s = std::max(3, ((int)(0.45 * r_lo) | 1));
307 cv::morphologyEx(dark, dark, cv::MORPH_OPEN,
308 cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(ko_s, ko_s)));
309 cv::morphologyEx(dark, dark, cv::MORPH_CLOSE,
310 cv::getStructuringElement(cv::MORPH_ELLIPSE, cv::Size(kc_s, kc_s)));
311
312 std::vector<std::vector<cv::Point>> cnts;
313 cv::findContours(dark, cnts, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
314 for (const auto &c : cnts) {
315 double area = cv::contourArea(c);
316 if (area < area_lo || area > area_hi || c.size() < 5)
317 continue;
318 cv::RotatedRect rr = cv::fitEllipse(c);
319 EllipseCand e;
320 e.cx = rr.center.x;
321 e.cy = rr.center.y;
322 e.rx = rr.size.width / 2.0;
323 e.ry = rr.size.height / 2.0;
324 e.angle = rr.angle;
325 e.coverage = (rr.size.width * rr.size.height > 0)
326 ? area / (M_PI * rr.size.width * rr.size.height / 4.0)
327 : 0.0;
328 if (std::min(e.rx, e.ry) < 0.85 * r_lo)
329 continue;
330 /* Skip blobs glued to the frame border. */
331 if (e.cx < 25 || e.cy < 25 || e.cx > w - 25 || e.cy > h - 25)
332 continue;
333 found.push_back(refine_disc(sm, e));
334 }
335 }
336
337 std::vector<EllipseCand> filtered;
338 for (const auto &e : found) {
339 double ratio = std::min(e.rx, e.ry) / std::max(e.rx, e.ry);
340 /* Streaky bottom discs often land near coverage ~0.3 after refine. */
341 if (e.coverage >= 0.30 && ratio > 0.55 && std::min(e.rx, e.ry) >= 0.8 * r_lo)
342 filtered.push_back(e);
343 }
344 /* Prefer high contrast, then better fill / circularity. */
345 std::sort(filtered.begin(), filtered.end(), [](const EllipseCand &a, const EllipseCand &b) {
346 double sa = a.score + 0.15 * a.coverage;
347 double sb = b.score + 0.15 * b.coverage;
348 return sa > sb;
349 });
350
351 std::vector<EllipseCand> kept;
352 for (const auto &e : filtered) {
353 bool hit = false;
354 for (const auto &k2 : kept) {
355 if (overlaps(e, k2)) {
356 hit = true;
357 break;
358 }
359 }
360 if (!hit)
361 kept.push_back(e);
362 if ((int)kept.size() >= nwant)
363 break;
364 }
365
366 /* Stable naming: top row left-to-right, then lower marks by y. */
367 std::sort(kept.begin(), kept.end(),
368 [](const EllipseCand &a, const EllipseCand &b) { return a.cy < b.cy; });
369 if (kept.size() >= 2) {
370 double ysep = kept[1].cy - kept[0].cy;
371 if (ysep < 0.35 * h && kept[0].cx > kept[1].cx)
372 std::swap(kept[0], kept[1]);
373 }
374
375 for (size_t i = 0; i < kept.size() && out->n < ROI_MAX; i++) {
376 EllipseCand e = kept[i];
377 /* Recenter on dark pixels so streaky discs do not leave the box on the rim. */
378 {
379 cv::Mat m = cv::Mat::zeros(sm.size(), CV_8U);
380 draw_ellipse_mask(m, e.cx, e.cy, e.rx, e.ry, e.angle, 0.85);
381 double sx = 0, sy = 0;
382 int n = 0;
383 for (int y = 0; y < h; y++) {
384 const uchar *mp = m.ptr<uchar>(y);
385 const uchar *sp = sm.ptr<uchar>(y);
386 for (int x = 0; x < w; x++) {
387 if (mp[x] && sp[x] < 110) {
388 sx += x;
389 sy += y;
390 n++;
391 }
392 }
393 }
394 if (n > 50) {
395 e.cx = sx / n;
396 e.cy = sy / n;
397 }
398 }
399 double r = std::min(e.rx, e.ry);
400 int side = std::max(1, (int)std::lround(r * M_SQRT2 * 0.50));
401 roi_t box;
402 std::memset(&box, 0, sizeof(box));
403 box.w = side+30;
404 box.h = side+30;
405 box.x = (int)std::lround(e.cx - side * 0.5);
406 box.y = (int)std::lround(e.cy - side * 0.5)-30;
407 box.pixels = box.w * box.h;
408 std::snprintf(box.name, sizeof(box.name), "ring%zu", i + 1);
409 out->item[out->n++] = box;
410 }
411 roi_clip(out, w, h);
412 LOG_INFO("detect_boxes_cv: kept %d mark(s)", out->n);
413 return 0;
414}
