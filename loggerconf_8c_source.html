98static void trim(char* s)
99{
100 size_t len;
101 int i;
102
103 if (s == NULL) {
104 return;
105 }
106 len = strlen(s);
107 if (len == 0) {
108 return;
109 }
110 /* trim right */
111 for (i = len - 1; i >= 0 && isspace(s[i]); i--) {}
112 s[i + 1] = '\0';
113 /* trim left */
114 for (i = 0; s[i] != '\0' && isspace(s[i]); i++) {}
115 memmove(s, &s[i], len - i);
116 s[len - i] = '\0';
