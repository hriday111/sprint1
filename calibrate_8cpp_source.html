40static void laser_on(void)
41{
42 int fd;
43
44 if ((fd = open(CMD_PIPE, O_WRONLY | O_NONBLOCK)) == -1) {
45 LOG_WARN("no command FIFO; assuming the laser is already on");
46 return;
47 }
48
49 const char cmd[] = "b255\n1001\n";
50
51 if (write(fd, cmd, sizeof(cmd) - 1) < 0 && errno != EAGAIN && errno != EPIPE) {
52 ERR("write");
53 }
54
55 close(fd);
56 usleep(30000);
57 LOG_INFO("Requested laser on via FIFO");
58}
