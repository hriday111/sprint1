168static int ueye_set_exposure_ms(cam_t *cam, double ms, double *actual)
169{
170 double lo = 0.0, hi = 0.0;
171 if (check(is_Exposure(cam->hcam, IS_EXPOSURE_CMD_GET_EXPOSURE_RANGE_MIN, &lo, sizeof(lo)),
172 "is_Exposure MIN") ||
173 check(is_Exposure(cam->hcam, IS_EXPOSURE_CMD_GET_EXPOSURE_RANGE_MAX, &hi, sizeof(hi)),
174 "is_Exposure MAX"))
175 return -1;
176 if (ms < lo)
177 ms = lo;
178 if (ms > hi)
179 ms = hi;
180
181 double applied = 0.0;
182 if (check(is_SetExposureTime(cam->hcam, ms, &applied), "is_SetExposureTime"))
183 return -1;
184 /* Allow one frame time so the new exposure is on the sensor. */
185 usleep(150000);
186 if (actual) {
187 if (ueye_get_exposure_ms(cam, actual) != 0)
188 *actual = applied;
189 }
190 LOG_INFO("Exposure set: requested %.3f ms, applied %.3f ms", ms,
191 actual ? *actual : applied);
192 return 0;
193}

195static int ueye_grab(cam_t *cam, unsigned char **pixels, int *width, int *height,
196 int *pitch)
197{
198 if (check(is_FreezeVideo(cam->hcam, IS_WAIT), "is_FreezeVideo"))
199 return -1;
200
201 int x = CAM_CROP_X;
202 int y = CAM_CROP_Y;
203 int w = CAM_CROP_W;
204 int h = CAM_CROP_H;
205 if (x + w > cam->width || y + h > cam->height) {
206 LOG_ERROR("cam_grab: crop (%d,%d %dx%d) outside %dx%d sensor", x, y, w, h,
207 cam->width, cam->height);
208 return -1;
209 }
210
211 /* View into the buffer — no copy. pitch stays the full row stride. */
212 *pixels = (unsigned char *)cam->mem + (size_t)y * (size_t)cam->pitch + (size_t)x;
213 *width = w;
214 *height = h;
215 *pitch = (int)cam->pitch;
216 LOG_TRACE("Grabbed crop %dx%d at (%d,%d)", w, h, x, y);
217 return 0;
218}
