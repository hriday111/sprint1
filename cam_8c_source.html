90 check(is_InquireImageMem(cam->hcam, cam->mem, cam->mem_id,

114 cam->open = 1;

141int cam_set_exposure_ms(cam_t *cam, double ms, double *actual)
142{
143 double lo = 0.0, hi = 0.0;
144 if (check(is_Exposure(cam->hcam, IS_EXPOSURE_CMD_GET_EXPOSURE_RANGE_MIN, &lo, sizeof(lo)),
145 "is_Exposure MIN") ||
146 check(is_Exposure(cam->hcam, IS_EXPOSURE_CMD_GET_EXPOSURE_RANGE_MAX, &hi, sizeof(hi)),
147 "is_Exposure MAX"))
148 return -1;
149 if (ms < lo)
150 ms = lo;
151 if (ms > hi)
152 ms = hi;
153
154 double applied = 0.0;
155 if (check(is_SetExposureTime(cam->hcam, ms, &applied), "is_SetExposureTime"))
156 return -1;
157 /* Allow one frame time so the new exposure is on the sensor. */
158 usleep(150000);
159 if (actual) {
160 if (cam_get_exposure_ms(cam, actual) != 0)
161 *actual = applied;
162 }
163 LOG_INFO("Exposure set: requested %.3f ms, applied %.3f ms", ms,
164 actual ? *actual : applied);
165 return 0;
166}

168int cam_grab(cam_t *cam, unsigned char **pixels, int *width, int *height, int *pitch)
169{
170 if (check(is_FreezeVideo(cam->hcam, IS_WAIT), "is_FreezeVideo"))
171 return -1;
172
173 int x = CAM_CROP_X;
174 int y = CAM_CROP_Y;
175 int w = CAM_CROP_W;
176 int h = CAM_CROP_H;
177 if ( x + w > cam->width || y + h > cam->height) {
178 LOG_ERROR("cam_grab: crop (%d,%d %dx%d) outside %dx%d sensor", x, y, w, h,
179 cam->width, cam->height);
180 return -1;
181 }
182
183 /* View into the buffer — no copy. pitch stays the full row stride. */
184 *pixels = (unsigned char *)cam->mem + (size_t)y * (size_t)cam->pitch + (size_t)x;
185 *width = w;
186 *height = h;
187 *pitch = (int)cam->pitch;
188 LOG_TRACE("Grabbed crop %dx%d at (%d,%d)", w, h, x, y);
189 return 0;
