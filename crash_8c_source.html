169void crash_setup(void)
170{
171 struct rlimit rl;
172 struct sigaction sa;
173 static const int fatals[] = {SIGSEGV, SIGABRT, SIGBUS, SIGFPE, SIGILL,
174 SIGQUIT, SIGSYS, SIGTRAP};
175 size_t i;
176
177 rl.rlim_cur = RLIM_INFINITY;
178 rl.rlim_max = RLIM_INFINITY;
179 if (setrlimit(RLIMIT_CORE, &rl) != 0)
180 LOG_WARN("setrlimit(RLIMIT_CORE) failed: %s (core dumps may be truncated)",
181 strerror(errno));
182
183 /* Ensure this process is dumpable even if it changed uid. */
184 if (prctl(PR_SET_DUMPABLE, 1L) != 0)
185 LOG_WARN("prctl(PR_SET_DUMPABLE) failed: %s", strerror(errno));
186
187 refresh_core_hint();
188 LOG_INFO("Core dumps enabled; expected path/hint: %s", g_core_hint);
189
190 memset(&sa, 0, sizeof(sa));
191 sa.sa_handler = crash_signal_handler;
192 sigemptyset(&sa.sa_mask);
193 sa.sa_flags = SA_RESETHAND;
194 for (i = 0; i < sizeof(fatals) / sizeof(fatals[0]); i++) {
195 if (sigaction(fatals[i], &sa, NULL) != 0)
196 LOG_WARN("sigaction(%s) failed: %s", signal_name(fatals[i]),
197 strerror(errno));
198 }
199}
